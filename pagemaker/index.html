<script>
// ===== 設定 =====
const PAGE_ID    = '';
const PAGE_TITLE = document.title;
const TOP_URL    = '../../index.html'; // トップページへの相対パス
const IMAGES = [


  


];
// ================

const gate    = document.getElementById('gate');
const content = document.getElementById('content');
const gateTitle = gate.querySelector('.gate-title');
const gateSub   = gate.querySelector('.gate-sub');

// ---- デバイス判定 ----
function detectDevice() {
  const ua = navigator.userAgent;
  if (/iPhone/.test(ua)) return 'iphone';
  if (navigator.maxTouchPoints > 1 && !/Macintosh/.test(ua)) return 'ipad';
  return 'mac';
}

// ---- iPad 分割ウィンドウ検出 ----
function checkIpadLayout() {
  const sw = screen.width, sh = screen.height;
  const isIpad16A = (sw === 1180 && sh === 820) || (sw === 820 && sh === 1180);
  if (!isIpad16A) return { ok: false, reason: '対応していないデバイスです' };
  const isSplit = window.innerWidth < screen.width * 0.9;
  if (isSplit) return { ok: false, reason: '全画面で開いてください' };
  return { ok: true };
}

// ---- レイアウトチェック（iPadのみ） ----
function layoutOk() {
  const device = detectDevice();
  if (device === 'iphone' || device === 'mac') return true;
  const result = checkIpadLayout();
  return result.ok;
}

function layoutMessage() {
  const result = checkIpadLayout();
  return result.reason || '全画面で開いてください';
}

// ---- 認証チェック ----
function isAuthed() {
  try {
    if (window.opener && window.opener.sessionStorage.getItem(`auth_${PAGE_ID}`) === '1') return true;
  } catch(_) {}
  return sessionStorage.getItem(`auth_${PAGE_ID}`) === '1';
}

// ---- Canvas で画像描画 ----
function drawImageToCanvas(src) {
  return new Promise(resolve => {
    const wrap = document.createElement('div');
    wrap.className = 'canvas-wrap';
    const canvas = document.createElement('canvas');
    const img = new Image();
    img.onload = () => {
      canvas.width  = img.naturalWidth;
      canvas.height = img.naturalHeight;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(img, 0, 0);
      wrap.appendChild(canvas);
      content.appendChild(wrap);
      resolve();
    };
    img.onerror = () => resolve();
    img.src = src;
  });
}

async function showContent() {
  document.title = PAGE_TITLE;
  sessionStorage.setItem(`auth_${PAGE_ID}`, '1');

  for (const filename of IMAGES) {
    await drawImageToCanvas(filename);
  }

  gate.style.display    = 'none';
  content.style.display = 'block';
}

function showGate(title, sub = '') {
  gateTitle.textContent = title;
  gateSub.textContent   = sub;
  gate.style.display    = 'flex';
  content.style.display = 'none';
}

function hideContent() {
  content.querySelectorAll('canvas').forEach(c => {
    const ctx = c.getContext('2d');
    ctx.clearRect(0, 0, c.width, c.height);
  });
  showGate('前面に戻すと再表示されます');
}

async function restoreContent() {
  content.querySelectorAll('.canvas-wrap').forEach(w => w.remove());
  for (const filename of IMAGES) {
    await drawImageToCanvas(filename);
  }
  gate.style.display    = 'none';
  content.style.display = 'block';
}

// ---- 初期表示 ----
(async () => {
  if (!isAuthed()) {
    showGate('認証されていません', 'トップページへ移動します...');
    setTimeout(() => { window.location.href = TOP_URL; }, 1500);
    return;
  }

  const device = detectDevice();
  if (device === 'ipad' && !layoutOk()) {
    showGate('全画面で開いてください', layoutMessage());
    window.addEventListener('resize', async () => {
      if (layoutOk()) {
        await showContent();
        setupVisibility();
      }
    });
    return;
  }

  await showContent();
  setupVisibility();
})();

// ---- Page Visibility API ----
function setupVisibility() {
  document.addEventListener('visibilitychange', async () => {
    if (!isAuthed()) return;
    if (document.hidden) {
      hideContent();
    } else {
      if (detectDevice() === 'ipad' && !layoutOk()) {
        showGate('全画面で開いてください', layoutMessage());
        return;
      }
      await restoreContent();
    }
  });

  if (detectDevice() === 'ipad') {
    window.addEventListener('resize', () => {
      if (!isAuthed()) return;
      if (!document.hidden && !layoutOk()) {
        hideContent();
        showGate('全画面で開いてください', layoutMessage());
      }
    });
  }
}

// ---- 右クリック禁止 ----
document.addEventListener('contextmenu', e => e.preventDefault());

// ---- キーボードショートカット無効化 ----
document.addEventListener('keydown', e => {
  if (e.ctrlKey && ['s','p','u','a'].includes(e.key.toLowerCase())) e.preventDefault();
  if (e.key === 'F12') e.preventDefault();
});
